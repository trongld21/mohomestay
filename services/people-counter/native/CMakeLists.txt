cmake_minimum_required(VERSION 3.20)
project(MoHomeImouBridge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "This SDK bridge requires Windows x64")
endif()
if(NOT EXISTS "${IMOU_SDK_ROOT}/include/lcopensdk_videoplay/LCOpenSDK_VideoPlay.h")
    message(FATAL_ERROR "Supply IMOU_SDK_ROOT from the official PC OpenSDK package")
endif()
find_package(Qt5 REQUIRED COMPONENTS Core Gui Widgets)
add_executable(mohome-imou-bridge bridge.cpp)
target_include_directories(mohome-imou-bridge PRIVATE "${IMOU_SDK_ROOT}/include")
target_link_directories(mohome-imou-bridge PRIVATE "${IMOU_SDK_ROOT}/x64")
target_compile_options(mohome-imou-bridge PRIVATE /utf-8)
target_compile_definitions(mohome-imou-bridge PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(mohome-imou-bridge PRIVATE Qt5::Core Qt5::Gui Qt5::Widgets LCOpenSDKmd user32 ws2_32)
